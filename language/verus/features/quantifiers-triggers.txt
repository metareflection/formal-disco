`forall|x: T| P(x)` and `exists|x: T| P(x)` are spec-level quantifier expressions. The SMT solver instantiates them via *triggers* — patterns of terms that, when matched, cause the solver to specialize the quantifier. Mark a sub-term with `#[trigger]` to choose the trigger explicitly.

```rust
use vstd::prelude::*;

verus! {

spec fn all_nonneg(s: Seq<int>) -> bool {
    forall|i: int| 0 <= i < s.len() ==> #[trigger] s[i] >= 0
}

spec fn has_zero(s: Seq<int>) -> bool {
    exists|i: int| 0 <= i < s.len() && s[i] == 0
}

proof fn nonneg_implies_no_neg(s: Seq<int>)
    requires all_nonneg(s),
    ensures forall|i: int| 0 <= i < s.len() ==> s[i] + 1 >= 1,
{
}

} // verus!
```
