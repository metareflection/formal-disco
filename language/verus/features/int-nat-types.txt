In specs and proofs Verus exposes mathematical `int` (unbounded signed) and `nat` (unbounded non-negative) types. They never overflow, so spec arithmetic is always exact. Convert between exec integer types and `int`/`nat` with `as int` / `as nat`.

```rust
use vstd::prelude::*;

verus! {

spec fn sum_to(n: nat) -> nat
    decreases n,
{
    if n == 0 { 0 } else { n + sum_to((n - 1) as nat) }
}

proof fn sum_formula(n: nat)
    ensures 2 * sum_to(n) == n * (n + 1),
    decreases n,
{
    if n == 0 {
    } else {
        sum_formula((n - 1) as nat);
        assert(n * (n + 1) == (n - 1) * n + 2 * n) by(nonlinear_arith);
    }
}

fn safe_double(x: u32) -> (r: u64)
    ensures r as int == 2 * (x as int),
{
    (x as u64) + (x as u64)
}

} // verus!
```
