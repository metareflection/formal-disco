In specs Verus extends Rust's logic with implication `==>`, equivalence `<==>`, and "big" conjunction `&&&` / disjunction `|||` for newline-friendly multi-line formulas. Chained inequalities like `0 <= x < 100` are also allowed in spec position.

```rust
use vstd::prelude::*;

verus! {

spec fn in_range(x: int, lo: int, hi: int) -> bool {
    lo <= x < hi
}

proof fn implication_demo(p: bool, q: bool)
    ensures (p ==> q) <==> (!p || q),
{
}

proof fn big_conjunction(x: int)
    requires
        0 <= x,
        x < 100,
        x % 2 == 0,
{
    assert({
        &&& 0 <= x
        &&& x % 2 == 0
        &&& {
            ||| x == 0
            ||| x >= 2
        }
    });
}

} // verus!
```
