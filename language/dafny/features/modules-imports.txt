Modules group declarations into a namespace. Members are accessed via dotted paths after `import`. `import opened M` brings M's members into scope unqualified. Modules can be nested, and abstract modules (declared with `abstract module`) can be refined by concrete modules.

```dafny
module Math {
  function Abs(x: int): int
  {
    if x < 0 then -x else x
  }

  lemma AbsNonNegative(x: int)
    ensures Abs(x) >= 0
  {
  }
}

module Geometry {
  import M = Math

  function ManhattanDistance(x1: int, y1: int, x2: int, y2: int): int
  {
    M.Abs(x1 - x2) + M.Abs(y1 - y2)
  }
}

module Client {
  import opened Math

  method UseImported()
  {
    AbsNonNegative(-7);
    assert Abs(-7) == 7;
  }
}
```
