Methods are imperative procedures with input parameters, named output parameters (`returns (...)`), and a body that may contain assignments, calls, and control flow. They specify behavior with `requires` (preconditions) and `ensures` (postconditions). Constructors initialize objects and use `new`. Methods can also have multiple out-parameters via parallel assignment.

```dafny
method MinMax(a: int, b: int) returns (lo: int, hi: int)
  ensures lo <= hi
  ensures lo == a || lo == b
  ensures hi == a || hi == b
  ensures {lo, hi} == {a, b}
{
  if a <= b {
    lo, hi := a, b;
  } else {
    lo, hi := b, a;
  }
}

method Swap<T>(a: T, b: T) returns (x: T, y: T)
  ensures x == b && y == a
{
  x, y := b, a;
}

method UseMethods()
{
  var lo, hi := MinMax(7, 3);
  assert lo == 3 && hi == 7;

  var p, q := Swap(1, 2);
  assert p == 2 && q == 1;
}
```
